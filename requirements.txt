# Full local/development dependencies for the pronunciation AI microservice.

# --- Web layer ---
fastapi
uvicorn[standard]
python-multipart          # multipart/form-data (audio uploads)
python-dotenv

# --- Acoustic model + audio ---
transformers
torch
torchaudio
librosa
soundfile
# DeepFilterNet 0.5.6 requires NumPy <2.
numpy==1.26.4
scipy
# DeepFilterLib publishes prebuilt Windows/Linux wheels through CPython 3.11.
# Use Python 3.11 (see .python-version) to avoid requiring a local Rust toolchain.
deepfilternet==0.5.6
silero-vad==6.2.1
cleanvoice-sdk==2.0.5       # optional external fallback; requires CLEANVOICE_API_KEY

# --- Scoring (PanPhon is REQUIRED for trusted scoring; without it the service
#     runs in the untrusted fallback and never updates mastery) ---
panphon

# --- Optional generation ---
openai                     # optional LLM exercise-generation fallback

# --- Reference G2P path ---
spacy                      # POS-aware heteronym resolution; also install en_core_web_sm
# NeMo is required at runtime and is installed separately through
# requirements-nemo.txt. This split keeps lightweight test installs possible;
# the service fails startup unless NeMo is present or G2P_REQUIRE_NEMO=0 was
# explicitly selected.

# --- Tests (the CI suite mocks the acoustic layer, so torch is not required
#     to run the mocked tests) ---
pytest
httpx                      # FastAPI TestClient transport
