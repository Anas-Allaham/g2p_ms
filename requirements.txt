# Full local/development dependencies for the pronunciation AI microservice.

# --- Web layer ---
fastapi
uvicorn[standard]
python-multipart          # multipart/form-data (audio uploads)
python-dotenv

# --- Acoustic model + audio ---
transformers
torch
librosa
soundfile
imageio-ffmpeg
numpy
noisereduce
scipy

# --- Scoring (PanPhon is REQUIRED for trusted scoring; without it the service
#     runs in the untrusted fallback and never updates mastery) ---
panphon

# --- Optional preprocessing / generation ---
cleanvoice-sdk>=2.0.5,<3   # optional server-side enhancement (needs a key)
openai                     # optional LLM exercise-generation fallback

# --- Preferred reference G2P path ---
spacy                      # POS-aware heteronym resolution; also install en_core_web_sm
nemo_toolkit[tts]          # NeMo IpaG2p fallback for non-heteronym words

# --- Tests (the CI suite mocks the acoustic layer, so torch is not required
#     to run the mocked tests) ---
pytest
httpx                      # FastAPI TestClient transport
